Part 1: The Strict <Image /> Component
Google penalizes pages heavily if images load slowly or if they push content down as they render. This component forces you (or your AI) to define dimensions and automatically handles lazy loading and fetch priorities.

Create this component in your kit: /lib/seo/Image.jsx

JavaScript
import React from 'react';

export const Image = ({ 
  src, 
  alt, 
  width, 
  height, 
  priority = false, 
  className = '',
  ...props 
}) => {
  // Enforce alt tags for accessibility and SEO
  if (!alt) {
    console.warn(`SEO Warning: Image with src "${src}" is missing an alt tag.`);
  }

  // Enforce explicit dimensions to prevent Cumulative Layout Shift (CLS)
  if (!width || !height) {
    console.warn(`SEO Warning: Image with src "${src}" is missing explicit width or height.`);
  }

  return (
    <img
      src={src}
      alt={alt || 'Image description missing'}
      width={width}
      height={height}
      className={className}
      // If priority is true, load it immediately for the Largest Contentful Paint (LCP)
      // Otherwise, defer it to save bandwidth and speed up initial render
      loading={priority ? 'eager' : 'lazy'}
      fetchpriority={priority ? 'high' : 'auto'}
      // Decode asynchronously to keep the main thread unblocked
      decoding="async"
      {...props}
    />
  );
};
